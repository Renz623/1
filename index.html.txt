    // 懒加载 + 压缩优化感知 (实际图片体积因 picsum 已优化, 同时在加载后标记)
    function lazyLoadImages() {
        const imageObserver = new IntersectionObserver((entries, observer) => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    const img = entry.target;
                    const src = img.getAttribute('data-src');
                    if (src) {
                        // 模拟图片体积压缩前2MB->200KB, 但真正实际网络请求已经是压缩资源，我们可加加载计时展示性能优化文案。
                        img.src = src;
                        img.onload = () => {
                            img.setAttribute('lazy', 'loaded');
                            img.style.opacity = '1';
                            loadedCount++;
                            if (loadedCount === artworks.length) {
                                console.log('[性能优化] 所有图片懒加载完毕，原图从默认压缩尺寸加载(约~200KB)');
                            }
                        };
                        img.removeAttribute('data-src');
                    }
                    observer.unobserve(img);
                }
            });
        }, { rootMargin: '200px' });
        document.querySelectorAll('.art-img').forEach(img => {
            imageObserver.observe(img);
        });
    }

    let messages = [];
    function loadMessages() {
        const stored = localStorage.getItem('art_gallery_messages');
        if(stored) {
            messages = JSON.parse(stored);
        } else {

    }
    function saveMessages() {
        localStorage.setItem('art_gallery_messages', JSON.stringify(messages));
    }
    function renderMessages() {
        const msgListDiv = document.getElementById('msgList');
        if(messages.length === 0) {
            msgListDiv.innerHTML = '<div style="text-align:center; padding:20px;">✨ 暂无留言，来做第一个留言的人吧 ✨</div>';
            return;
        }
        msgListDiv.innerHTML = messages.slice().reverse().map(msg => `
            <div class="msg-item">
                <div class="msg-name">${escapeHtml(msg.name)}</div>
                <div class="msg-text">${escapeHtml(msg.text)}</div>
                <div class="msg-date">${msg.date}</div>
            </div>
        `).join('');
    }
    function escapeHtml(str) {
        if(!str) return '';
        return str.replace(/[&<>]/g, function(m) {
            if(m === '&') return '&amp;';
            if(m === '<') return '&lt;';
            if(m === '>') return '&gt;';
            return m;
        }).replace(/[\uD800-\uDBFF][\uDC00-\uDFFF]/g, function(c) {
            return c;
        });
    }
    function addMessage(name, text) {

            text: text.trim().substring(0, 200),
            date: new Date().toLocaleString()
        };
        messages.push(newMsg);
        saveMessages();
        renderMessages();
        return true;
    }
